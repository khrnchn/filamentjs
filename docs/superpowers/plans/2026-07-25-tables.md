# FilamentJS Tables Package Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `@filamentjs/tables`: fluent column, filter, and action builders; a `buildTable` normalizer; column-state resolution (dot-path or accessor); and a serializable table spec for the client. Plus the `TableParams` / `TableResult` contract types. Pure logic, no DB (the Drizzle query resolver is a later plan).

**Architecture:** Tables builds on `@filamentjs/core` (reuses `getPath` for dot-path column values). Columns/filters/actions are plain builders that accumulate config objects; `buildTable` turns an input of builders into a `TableConfig` of plain nodes. `resolveColumnValue` reads a row cell by walking the dot path or running an accessor. `resolveTableSpec` emits a JSON-serializable description (functions stripped, default labels filled) for the client renderer. The `TableParams -> TableResult` contract is the server-driven data boundary that the future Drizzle resolver will satisfy.

**Tech Stack:** TypeScript (strict), Vitest, tsup. Depends on `@filamentjs/core`.

## Global Constraints

- Package manager: **pnpm** with workspaces. Node >= 20.
- TypeScript **strict** mode. No `any` in public signatures; `unknown` at boundaries.
- No em dashes in comments, docs, or commit messages.
- Comments explain gotchas/constraints only. No verbose docstrings.
- Builds with **tsup** to ESM, exports types.
- Tests colocated as `*.test.ts`, run by Vitest.
- Package name: `@filamentjs/tables`.
- Consumes core's public API only: `getPath`, and types `SchemaNode` (not needed here beyond reference). Tables do NOT reuse `NodeBuilder`; columns are a distinct concept with their own builder.

// packages/tables/src/factory.ts
import { BadgeColumnBuilder, ColumnBuilder } from './columns.js';
import { FilterBuilder } from './filters.js';
import { ActionBuilder } from './actions.js';
import type { TableConfig } from './types.js';

export const t = {
  text: (name: string) => new ColumnBuilder('text', name),
  badge: (name: string) => new BadgeColumnBuilder('badge', name),
  boolean: (name: string) => new ColumnBuilder('boolean', name),
  icon: (name: string) => new ColumnBuilder('icon', name),
  image: (name: string) => new ColumnBuilder('image', name),
  select: (name: string) => new FilterBuilder('select', name),
  ternary: (name: string) => new FilterBuilder('ternary', name),
  editAction: () => new ActionBuilder({ type: 'edit', name: 'edit' }),
  deleteAction: () =>
    new ActionBuilder({ type: 'delete', name: 'delete', requiresConfirmation: true, destructive: true }),
  createAction: () => new ActionBuilder({ type: 'create', name: 'create' }),
  deleteBulkAction: () => new ActionBuilder({ type: 'deleteBulk', name: 'deleteBulk', destructive: true }),
};

export interface TableConfigInput {
  columns: ColumnBuilder[];
  filters?: FilterBuilder[];
  actions?: ActionBuilder[];
  bulkActions?: ActionBuilder[];
  headerActions?: ActionBuilder[];
}

export function buildTable(input: TableConfigInput): TableConfig {
  return {
    columns: input.columns.map((c) => c.build()),
    filters: (input.filters ?? []).map((f) => f.build()),
    actions: (input.actions ?? []).map((a) => a.build()),
    bulkActions: (input.bulkActions ?? []).map((a) => a.build()),
    headerActions: (input.headerActions ?? []).map((a) => a.build()),
  };
}

// packages/tables/src/resolve-value.ts
import { getPath } from '@filamentjs/core';
import type { ColumnNode, Row } from './types.js';

export function resolveColumnValue(row: Row, column: ColumnNode): unknown {
  if (column.accessor) return column.accessor(row);
  return getPath(row, column.name);
}

export function humanizeLabel(name: string): string {
  const last = name.split('.').pop() ?? name;
  return last
    .replace(/([a-z0-9])([A-Z])/g, '$1 $2')
    .replace(/[_-]+/g, ' ')
    .trim()
    .split(/\s+/)
    .map((w) => w.charAt(0).toUpperCase() + w.slice(1))
    .join(' ');
}
```

// packages/tables/src/resolve-spec.ts
import { humanizeLabel } from './resolve-value.js';
import type { ActionNode, ColumnNode, FilterNode, TableConfig } from './types.js';

export interface ColumnSpec {
  type: string;
  name: string;
  label: string;
  sortable: boolean;
  searchable: boolean;
  colors?: Record<string, string>;
}

export interface FilterSpec {
  type: string;
  name: string;
  label: string;
  options?: Record<string, string>;
  multiple?: boolean;
}

export interface TableSpec {
  columns: ColumnSpec[];
  filters: FilterSpec[];
  actions: ActionNode[];
  bulkActions: ActionNode[];
  headerActions: ActionNode[];
}

function columnSpec(col: ColumnNode): ColumnSpec {
  const spec: ColumnSpec = {
    type: col.type,
    name: col.name,
    label: col.label ?? humanizeLabel(col.name),
    sortable: col.sortable ?? false,
    searchable: col.searchable ?? false,
  };
  if (col.colors) spec.colors = col.colors;
  return spec;
}

function filterSpec(filter: FilterNode): FilterSpec {
  const spec: FilterSpec = {
    type: filter.type,
    name: filter.name,
    label: filter.label ?? humanizeLabel(filter.name),
  };
  if (filter.options) spec.options = filter.options;
  if (filter.multiple) spec.multiple = true;
  return spec;
}

export function resolveTableSpec(config: TableConfig): TableSpec {
  return {
    columns: config.columns.map(columnSpec),
    filters: config.filters.map(filterSpec),
    actions: config.actions,
    bulkActions: config.bulkActions,
    headerActions: config.headerActions,
  };
}
```
